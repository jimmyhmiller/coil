;;; Compact adjacency lists indexed by positive instruction reference.
;;; Each use records its consuming instruction and one-based operand position.
(module stogir.uses)
(import "stogir.graph" :as g)
(import "stogir.node" :as n)
(import "stogir.opcode" :as op)
(import "coil.arraylist" :use [ArrayList])
(import "coil.alloc" :use [Allocator])
(import "coil.result" :use [Some None])

(defstruct Use [(user i32) (operand i32)])

(defstruct Range [(offset i64) (count i64)])

(defstruct Uses [(ranges (ArrayList Range)) (edges (ArrayList Use))])

(defn edge-kind? [(kind (slice u8))] (-> bool)
  (or (= kind "def")
      (= kind "ref")
      (= kind "var")
      (= kind "src")
      (= kind "reg")
      (= kind "grd")
      (= kind "ret")
      (= kind "lbl")))

;;; Numeric form of the same public reference-edge categories. Codes come
;;; directly from opcode's packed operand metadata; no string dispatch is needed.
;;; Mask members: def, ref, var, src, reg, grd, ret, lbl.
(defn edge-code? [(kind u32)] (-> bool)
  (and (< kind 15) (!= (& (: 24092 u32) (<< (: 1 u32) kind)) 0)))

;;; Requires structural verification first. Constants are excluded.
;;; Two graph walks count and then fill contiguous per-definition ranges.
(defn build [(graph g/Graph) (allocator (dyn Allocator))] (-> Uses)
  (let [(mut result)
          (Uses :ranges (ArrayList::new [Range] allocator)
                :edges (ArrayList::new [Use] allocator))
        (mut r) 1]
    (extend-zeroed! (mut (.ranges result)) (len (.instructions graph)))
    ; Upstream uses two explicit walks. Count only in this walk; no dynamic
    ; count/fill selection or fill cursor is needed at each reference edge.
    (while (< r (len (.instructions graph)))
      (let [node (get (.instructions graph) r)
            count (get (.arities graph) r)
            opcode (cast i32 (n/opcode node))
            facts (if (op/valid? opcode) (get (op/metadata) (cast i64 opcode)) 0)
            (mut i) (: 1 i32)]
        (while (<= i count)
          (when (edge-code? (op/operand-code-word facts i))
            (match (g/operand graph (cast i32 r) i)
              (Some [value]
                    (when (> value 0)
                      (let [idx (cast i64 value)
                            (mut range) (get (.ranges result) idx)]
                        (set! (.count range) (+ (.count range) 1))
                        (set! (mut (.ranges result)) idx range))))
              (None [] 0)))
          (set! i (+ i 1)))
        (set! r (+ r (/ (+ (cast i64 count) 4) 4)))))
    (let [(mut offset) 0]
      (for [i 0 (len (.ranges result))]
        (let [(mut range) (get (.ranges result) i)
              count (.count range)]
          (set! (.offset range) offset)
          ; Counts become per-definition fill cursors, as in ir.c. The same
          ; immutable input walk restores their final values exactly.
          (set! (.count range) 0)
          (set! (mut (.ranges result)) i range)
          (set! offset (+ offset count))))
      (extend-zeroed! (mut (.edges result)) offset))
    (set! r 1)
    (while (< r (len (.instructions graph)))
      (let [node (get (.instructions graph) r)
            count (get (.arities graph) r)
            opcode (cast i32 (n/opcode node))
            facts (if (op/valid? opcode) (get (op/metadata) (cast i64 opcode)) 0)
            (mut i) (: 1 i32)]
        (while (<= i count)
          (when (edge-code? (op/operand-code-word facts i))
            (match (g/operand graph (cast i32 r) i)
              (Some [value]
                    (when (> value 0)
                      (let [idx (cast i64 value)
                            (mut range) (get (.ranges result) idx)]
                        (set! (mut (.edges result)) (+ (.offset range) (.count range))
                              (Use :user (cast i32 r) :operand i))
                        (set! (.count range) (+ (.count range) 1))
                        (set! (mut (.ranges result)) idx range))))
              (None [] 0)))
          (set! i (+ i 1)))
        (set! r (+ r (/ (+ (cast i64 count) 4) 4)))))
    result))

(defn release! [(uses (mut Uses))] (-> void)
  (free! (mut (.ranges uses)))
  (free! (mut (.edges uses))))

;;; Mutable lists follow ir.c's ir_use_list_add/remove_one policy. References and
;;; operand positions identify edges. Re-read ranges after mutation; no interior
;;; pointer may survive an add, which may move a bucket or grow the edge buffer.
(defn add! [(lists (mut Uses)) (definition i32) (edge Use)] (-> void)
  (let [index (cast i64 definition)
        (mut range) (get (.ranges lists) index)
        end (+ (.offset range) (.count range))]
    (if (and (< end (len (.edges lists)))
             (= (.user (get (.edges lists) end)) 0))
        (set! (mut (.edges lists)) end edge)
        (if (= end (len (.edges lists)))
            (push! (mut (.edges lists)) edge)
            (let [start (len (.edges lists))]
              (for [i 0 (.count range)]
                (let [copy (get (.edges lists) (+ (.offset range) i))]
                  (push! (mut (.edges lists)) copy)))
              (push! (mut (.edges lists)) edge)
              (set! (.offset range) start))))
    (set! (.count range) (+ (.count range) 1))
    (set! (mut (.ranges lists)) index range)))

;;; Repeated operands are separate edges. Remove exactly the requested position,
;;; retaining order and exposing the final slot as reusable zero-filled space.
(defn remove! [(lists (mut Uses)) (definition i32) (edge Use)] (-> bool)
  (let [index (cast i64 definition)
        (mut range) (get (.ranges lists) index)
        end (+ (.offset range) (.count range))
        (mut found) (: -1 i64)]
    (for [i (.offset range) end]
      (let [candidate (get (.edges lists) i)]
        (when (and (< found 0)
                   (= (.user candidate) (.user edge))
                   (= (.operand candidate) (.operand edge)))
          (set! found i))))
    (when (>= found 0)
      (for [i found (- end 1)]
        (set! (mut (.edges lists)) i (get (.edges lists) (+ i 1))))
      (set! (mut (.edges lists)) (- end 1) (Use :user 0 :operand 0))
      (set! (.count range) (- (.count range) 1))
      (set! (mut (.ranges lists)) index range))
    (>= found 0)))

(defn replace! [(lists (mut Uses)) (definition i32) (old Use) (new Use)]
  (-> bool)
  (let [range (get (.ranges lists) (cast i64 definition))
        (mut found) false]
    (for [i (.offset range) (+ (.offset range) (.count range))]
      (let [candidate (get (.edges lists) i)]
        (when (and (not found)
                   (= (.user candidate) (.user old))
                   (= (.operand candidate) (.operand old)))
          (set! (mut (.edges lists)) i new)
          (set! found true))))
    found))
